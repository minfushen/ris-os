### 2.2 数据模型设计

#### 企业档案表 (enterprises)
```sql
CREATE TABLE enterprises (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    company_name TEXT NOT NULL UNIQUE,
    credit_code TEXT,  -- 统一社会信用代码
    legal_person TEXT,  -- 法定代表人
    registered_capital TEXT,  -- 注册资本
    established_date TEXT,  -- 成立日期

    registered_address TEXT,  -- 注册地址
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 风险评估记录表 (risk_assessments)
```sql
CREATE TABLE risk_assessments (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    enterprise_id INTEGER NOT NULL,
    overall_risk TEXT NOT NULL,  -- CRITICAL/HIGH/MEDIUM/LOW
    assessment_data TEXT,  -- JSON 格式的完整评估数据
    assessment_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (enterprise_id) REFERENCES enterprises(id)
);
```

#### 预警记录表 (alerts)
```sql
CREATE TABLE alerts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    enterprise_id INTEGER NOT NULL,
    alert_type TEXT NOT NULL,  -- 预警类型
    alert_level TEXT NOT NULL,  -- CRITICAL/HIGH/MEDIUM/LOW
    alert_source TEXT,  -- 预警来源
    alert_status TEXT DEFAULT 'active',  -- active/resolved/ignored
    triggered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    resolved_at TIMESTAMP,
    FOREIGN KEY (enterprise_id) REFERENCES enterprises(id)
);
```

#### 客户关联表 (customers)
```sql
CREATE TABLE customers (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    enterprise_id INTEGER NOT NULL,
    customer_name TEXT NOT NULL,  -- 客户名称
    customer_type TEXT,  -- 客户类型
    loan_amount REAL,  -- 贷款金额
    loan_status TEXT,  -- 贷款状态
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (enterprise_id) REFERENCES enterprises(id)

class AlertManager:
    """预警管理器"""
    
    async def check_and_trigger_alerts(self, enterprise_id: int, assessment: dict):
        """检查并触发预警"""
        # 1. 根据风险评估结果判断是否需要预警
        if assessment['overall_risk'] in ['CRITICAL', 'HIGH']:
            # 2. 创建预警记录
            alert = await self.create_alert(
                enterprise_id=enterprise_id,
                alert_type='risk_assessment',
                alert_level=assessment['overall_risk'],
                alert_source='企业风险评估 Agent'
            )
            # 3. 发送通知（可选）
            await self.send_notification(alert)
        
        return alert
```

## 六、技术要点

### 6.1 数据缓存策略

**内存缓存**：
```python
from functools import lru_cache
from datetime import datetime, timedelta

class CacheManager:
    def __init__(self):
        self._cache = {}
        self._cache_ttl = timedelta(hours=1)
    
    def get(self, key: str):
        if key in self._cache:
            value, timestamp = self._cache[key]
            if datetime.now() - timestamp < self._cache_ttl:
                return value
        return None
    
    def set(self, key: str, value):
        self._cache[key] = (value, datetime.now())
```

### 6.2 并发控制

**批量处理**：
```python
import asyncio

async def batch_assess_enterprises(enterprise_ids: list):
    """批量评估企业风险"""
    tasks = [
        assess_enterprise_risk(eid)
        for eid in enterprise_ids
    ]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    return results
```
